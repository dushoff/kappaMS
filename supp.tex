\section{Analytical Results} \label{sec_proof}
The standard SIR model is described by the following equations:
%\begin{equation}\label{eq:SIR}
\begin{align*}
        \dot{x}(t) &= - \Ro x(t)y(t), \\
        \dot{y}(t) &= \Ro x(t)y(t) - y(t),
\end{align*}
%\end{equation}
where $x(t)$ and $y(t)$ denote the proportion of susceptible and infectious individuals, respectively. 
Time has been normalized by the recovery rate.
Let $i(t)$ denote the incidence $\Ro x(t)y(t)$, and define the size of the epidemic as follows:
\[Z = \int_{0}^\infty dt i(t).\]

Let $f(\cdot)$ denote the probability density function of the infectious period. Define $F(t)$ as the corresponding survival distribution, that is
\begin{equation} \label{eq:survial}
    F(t) = 1 - \int_0^t d\tau f(\tau) = \int_t^\infty d\tau f(\tau).
\end{equation}

Note that $F(t)$ is the probability that an individual infected at time $0$ remains infectious after time $t$. 
%Note also that in changing the limits of the integral, we only used the fact that $f$ is a normalized probability density function -- we did not use the fact that $f$ is an exponential.


We define the \emph{expected case reproductive number} between time $\tau$ and $\rho$ as 
\[C(\tau,\rho) = \Ro \int_\tau^\rho dt\; x(t).\]
The $k^\text{th}$ weighted raw moment of $C$ is then defined as
\begin{equation}
    \label{eq:Moments}
    C_k = \int_0^\infty d\tau \int_\tau^\infty d\rho \; w(\tau,\rho) (C(\tau,\rho))^k,
\end{equation}
where $w(\tau,\rho) = i(\tau)f(\rho-\tau)$ is the size of the cohort that became infectious at time $\tau$ and recovered at time $\rho$.

In Section~\ref{proof_timeInf_initial0}, we prove the following statement:
The excess variation of the individual case reproductive number in the standard SIR model outbreak is 1, 
provided that the initially infectious cohort size is negligible.
Mathematically, this amounts to proving that

\begin{align}\label{eq:target}
\kappa= \frac{C_0C_2}{C_1^2} - 1 = 1.
\end{align}




\subsection{Proof} \label{proof_timeInf_initial0}
We will calculate $C_0$, $C_1$ and $C_2$. We can calculate $C_0$ and $C_1$ without making any assumptions about the functional form of $f$ (we don't have to assume that the wait times are exponentially distributed). We do need $F$ to be memoryless, i.e., obey $F(a+b) = F(a)F(b)$, to get the ``right'' value of $C_2$.
Calculating each of these moments essentially involves multiple manipulations of the limits of the integrals involved. 
To proceed, we assume that  size of the initial infectious cohort is negligible.

%\subsubsection{Proof of $C_0 = Z$} \label{sec:C0}
From \autoref{eq:Moments}, we have
\begin{align*}
        C_0 &= \int_0^\infty d\tau \int_\tau^\infty d\rho \; w(\tau,\rho) \\
        &= \int_0^\infty d\tau \int_\tau^\infty d\rho \; i(\tau) f(\rho - \tau)
\end{align*}

Substituting $\omega + \tau $ for $\rho$ in the inner integral, results in

\begin{align*}
        C_0 &=\int_0^\infty d\tau \;i(\tau) \int_0^\infty d\omega \; f(\omega).
\end{align*}
Since $f$ is the residence time distribution, the inner integral is simply 1.
It follows that
\begin{align*}
 %   \begin{split} \label{eq:C0calc}
        C_0 &=\int_0^\infty d\tau \;i(\tau) \\
        &= Z. 
  %  \end{split}
\end{align*}

%\subsubsection{Proof of $C_1 = Z$}\label{sec:C1}

The next step is to prove that $C_1 = Z$.
From \autoref{eq:Moments}, we have
\begin{align*}
        C_1 &= \int_0^\infty d\tau \int_\tau^\infty d\rho \; w(\tau,\rho) C(\tau,\rho)\\
        &= \Ro\int_0^\infty d\tau \int_\tau^\infty d\rho i(\tau) f(\rho-\tau) \int_\tau^{\rho} dt x(t).
\end{align*}
Instead of first integrating over $t$ from $\tau$ to $\rho$ and then over $\rho$ from $\tau$ to $\infty$, we can first integrate over $\rho$ from $t$ to $\infty$ and then integrate over $t$ from $\tau$ to $\infty$, i.e.,
\begin{equation*}
    C_1 =\Ro\int_0^\infty d\tau \int_\tau^\infty dt i(\tau) x(t) \int_t^\infty d\rho f(\rho-\tau). 
\end{equation*}
Substitute $\rho = \omega + \tau$ in the innermost integral and using \autoref{eq:survial}, we get
\begin{align*}
    \begin{split}
        C_1 &= \Ro\int_0^\infty d\tau \int_\tau^\infty dt\; i(\tau)x(t) \int_{t-\tau}^\infty d\omega f(\omega)\\
        &=  \Ro\int_0^\infty d\tau \int_\tau^\infty dt\; i(\tau)x(t) F(t-\tau).
    \end{split}
\end{align*}
Note again that we have not used the fact that $f$ is exponential, we have merely used the definition of $F$ as an integral of $f$.
By interchanging the order of the integrals once more, we have
\begin{align*}
        C_1 &= \Ro\int_0^\infty d\tau \int_\tau^\infty dt\; i(\tau)x(t) \int_{t-\tau}^\infty d\omega f(\omega)\\
        &=  \Ro\int_0^\infty dt\; x(t) \int_0^t d\tau\; i(\tau)F(t-\tau). 
\end{align*}
Note that the inner integrand counts the number of individuals that entered the infectious state at some time $\tau < t$ and remain infectious  at time $t$. When we integrate over all times $\tau < t$, we simply get the number of individuals who are infections at time $t$, i.e., $y(t)$.
 So,
\begin{align}
        C_1 &= \Ro\int_0^\infty dt\; x(t) \int_0^t d\tau\; i(\tau)F(t-\tau) \label{eq:C1calc_prev}\\
        &= \Ro\int_0^\infty dt\; x(t)y(t)     \label{eq:C1calc}\\
        &= \int_0^\infty dt\; i(t)\\
        &=Z.
\end{align}
%\subsubsection{Proof of $C_2 = 2Z$}\label{sec:C2}
The next step is to show $C_2 = 2Z$.
Given \autoref{eq:Moments}, we have
\begin{align*}
        C_2 &= \int_0^\infty d\tau \int_\tau^\infty d\rho \; w(\tau,\rho) \left(C(\tau,\rho)\right)^2\\
        &= \Ro^2 \int_0^\infty d\tau \int_\tau^\infty d\rho \;i(\tau) f(\rho-\tau) \left(\int_\tau^{\rho} dt x(t)\right)^2
\end{align*}
Now we break down the square integral into a double integral
\begin{align*}
        \left(\int_\tau^\rho x(t)dt\right)^2 &= \left( \int_\tau^\rho x(t) dt\right) \left( \int_\tau^\rho x(s)ds\right)\\
        &=  \left( \int_\tau^\rho x(t) dt\right) \left( \int_\tau^\rho x(s)ds\right)\\
        &=  \int_\tau^\rho \int_\tau^\rho dt ds \; x(t) x(s).
\end{align*}
In $t-s$ space, this is integrating $x(t)x(s)$ over a square region whose vertices are $(\tau,\tau)$, $(\tau,\rho)$, $(\rho,\tau)$ and $(\rho,\rho)$. This is twice the integral over the triangle whose vertices are $(\tau,\tau)$, $(\tau,\rho)$ and $(\rho,\tau)$. So,
\begin{equation*}
        \left(\int_\tau^\rho x(t)dt\right)^2 = 2 \int_\tau^\rho ds \int_\tau^s dt \; x(t)x(s)
\end{equation*}
Plugging this back into the expression for $C_2$ yields
%\begin{align} \label{eq:SquareToTriangle}
    \begin{equation*}
        C_2 = 2\Ro^2 \int_0^\infty d\tau \int_\tau^\infty d\rho \int_\tau^\rho ds \int_\tau^s dt \; i(\tau) f(\rho-\tau) x(t)x(s).
    \end{equation*}
%\end{align}
First, we reorder the integrals over $\rho$ and $s$ and change the limits appropriately
\begin{align*}
    \begin{split}
        C_2 &= 2\Ro^2 \int_0^\infty d\tau \; i(\tau) \int_\tau^\infty ds \; x(s)\int_s^\infty d\rho \int_\tau^s dt  f(\rho-\tau) x(t).
    \end{split}
\end{align*}
Next, we reorder the integrals over $\rho$ and $t$ to get
\begin{align*}
    \begin{split}
        C_2 &= 2\Ro^2 \int_0^\infty d\tau \; i(\tau) \int_\tau^\infty ds \; x(s) \int_\tau^s dt \; x(t)\int_s^\infty d\rho f(\rho-\tau). 
    \end{split}
\end{align*}
By substituting $\rho = \omega + \tau$ and using \autoref{eq:survial}, we get
\begin{align*}
    \begin{split}
        C_2 &= 2\Ro^2 \int_0^\infty d\tau \; i(\tau) \int_\tau^\infty ds \; x(s) \int_\tau^s dt \; x(t)F(s-\tau).
    \end{split}
\end{align*}
We then reorder the integrals over $\tau$ and $s$
\begin{align*}
    \begin{split}
        C_2 &= 2\Ro^2 \int_0^\infty ds\;x(s)\int_0^s d\tau \int_\tau^s dt\; x(t)i(\tau)F(s-\tau).
    \end{split}
\end{align*}
 Swapping the integrals over $\tau$ and $t$, results in
\begin{align*}
  %  \begin{split} 
        C_2 = 2\Ro^2 \int_0^\infty ds\;x(s)\int_0^s dt \; x(t) \int_0^t d\tau\;i(\tau)F(s-\tau).
 %   \end{split}
\end{align*}
Now, we use the fact that the survival function $F$ is memoryless, i.e., $F(a+b) = F(a)F(b)$, and replace $F(s-\tau)$ with $F(s-t)F(t-\tau)$
%\begin{align} \label{eq:NeedMarkov}
    \begin{equation*}
        C_2 = 2\Ro^2 \int_0^\infty ds\;x(s)\int_0^s dt \; x(t) \int_0^t d\tau\;i(\tau)F(s-t)F(t-\tau).
    \end{equation*}
%\end{align}
Next, we use the same argument we used to go from \autoref{eq:C1calc_prev} to \autoref{eq:C1calc}, which results in
\begin{align*}
        C_2 &= 2\Ro^2 \int_0^\infty ds\;x(s)\int_0^s dt \; x(t) F(s-t)y(t)\\
        &= 2\Ro \int_0^\infty ds\;x(s)\int_0^s dt \; i(t) F(s-t)\\
        &= 2\Ro \int_0^\infty ds\;x(s)y(s)\\
        &= 2 \int_0^\infty ds\;i(s)\\
        &= 2Z.
\end{align*}

Finally, replacing $C_0$, $C_1$, and $C_2$ by $Z$, $Z$, and $2Z$, respectively, in \autoref{eq:target}, results $\kappa = 1$.

\subsection{Incorporating the Size of Initial Infections}
 \eqref{eq:target} 
was based on the assumption that the population who were infectious initially  was negligible.
In what follows, we revisit the results while relaxing this assumption.
 To account for the contribution of these initial infections, we assume that the epidemic starts abruptly at time $t=0$, i.e.,
 \begin{equation} \label{eq_incidence_dirac}
 	i(t)=y_0\delta(t), \quad t\in(-\infty,0].
 \end{equation}
Here, $y_0$ denotes the proportion of the population that were infectious at the beginning of the outbreak,  $ t = 0$, and $\delta(t)$ denotes the Dirac delta distribution \cite{Champredon2018}.
 Then, accounting for the initially infectious cohort, the proportion of the infected population that is infectious is given by
$$
y(t) = \int_{0^-}^t d\tau\; i(\tau)F(t-\tau).
$$

Here, the lower limit of the integral is $0^-$ to include the Dirac delta function at $t= 0$.
The total weight $C_0$ will now includes the initially infectious cohort size as well, i.e,
\begin{align*}
	C_0 = \int_{0^-}^{\infty} dt \; i(t)
\end{align*}
The first raw moment $C_1$ is given by
\begin{align*}
	C_1 &= \int_{0^-}^\infty d\tau \int_\tau^\infty d\rho \; w(\tau,\rho) C(\tau,\rho)\\
	& =\Ro\int_{0^-}^\infty d\tau \int_\tau^\infty dt i(\tau) x(t) \int_t^\infty d\rho f(\rho-\tau)\\
		&=  \Ro\int_{0^-}^\infty d\tau \int_\tau^\infty dt i(\tau)x(t) F(t-\tau)\\
	&=  \Ro\int_0^\infty dt x(t) \int_{0^-}^t d\tau i(\tau)F(t-\tau)\\
	 &= \Ro\int_{0^-}^\infty dt\; x(t) y(t).
\end{align*}
Given both $x(t)$ and $y(t)$ are bounded,
without loss of generality, the lower limit can be replaced with $0^+$. Thus,
the first raw moment $C_1$ becomes
\begin{align*}
C_1	&= \Ro\int_{0^+}^\infty dt\; x(t) y(t) \\
&=
\int_{0^+}^\infty dt\; i(t).
\end{align*}
We note that $C_1$ does not include the size of initially infectious cohort. 
Now we move on to recalculate  $C_2$, while taking the initially infectious cohort into account.
We have
\begin{align*}
	C_2 &= \int_{0^-}^\infty d\tau \int_\tau^\infty d\rho \; w(\tau,\rho) \left(C(\tau,\rho)\right)^2\\
	&= \Ro^2 \int_{0^-}^\infty d\tau \int_\tau^\infty d\rho \;i(\tau) f(\rho-\tau) \left(\int_\tau^{\rho} dt x(t)\right)^2 \\
\end{align*}
Using the same mathematical manipulation used in section~\ref{sec_proof}, we have
\begin{align*}
	C_2	& = 2\Ro^2 \int_{0^-}^\infty ds\;x(s)\int_{0^-}^s dt \; x(t) \int_{0^-}^t d\tau\;i(\tau)F(s-t)F(t-\tau)\\
	& = 2\Ro^2 \int_{0^-}^\infty ds\;x(s)\int_{0^-}^s dt \; x(t) F(s-t)y(t)
\end{align*}
We split the innermost integral into two integrals
 \begin{align*}
 	C_2 & = 2\Ro^2 \int_{0^-}^\infty ds\;x(s)\int_{0^-}^s dt \; x(t) F(s-t)y(t) \\
 	& = 2\Ro^2 \int_{0^-}^\infty ds\;x(s) \left( \underbrace{\int_{0^-}^{0^+} dt \; x(t) F(s-t)y(t)}_{0} + {\int_{0^+}^s dt \;  x(t) F(s-t)y(t)} \right)\\
 	& = 2\Ro^2 \int_{0^-}^\infty ds\;x(s) \int_{0^+}^s dt\; x(t)y(t) F(s-t) \\
 		& = 2\Ro \int_{0^-}^\infty ds\;x(s) \int_{0^+}^s dt\; i(t)F(s-t).
 \end{align*}
The innermost integral gives the  proportion of the population who entered the infectious state at some time $t>0$ and remain infectious at time $s$. Hence, it equals the total proportion infectious  at time $s$ minus the proportion of the population who were initially infectious and remain infectious at time $s$, i.e., $y(s) - y_0F(s).$ Having replaced the innermost integral with $y(s) - y_0F(s)$, the second raw moment $C_2$ becomes
 \begin{align*}
	C_2 & = 2\Ro \int_{0^-}^\infty ds\;x(s) \left( y(s) - y_0F(s) \right) \\
	& = 2\Ro \int_{0^-}^\infty ds\;x(s)y(s) -2\Ro \int_{0^-}^\infty ds\;x(s) y_0F(s) \\
		& = 2 \int_{0^+}^\infty i(s)ds -2\Ro \int_{0^+}^\infty ds\;x(s) y_0F(s)  \\
		& = 2C_1 - 2\Ro \int_{0^+}^\infty ds\;x(s) y_0F(s).
\end{align*}
Note that changing the lower limit of integration from $0^-$ to $0^+$ does not affect the result because the integrand is bounded.
The second term in the left hand side in the twice the expected infectiousness of the initial cohort.
Here we notice a couple of points. The second raw moment is $\emph{dependent}$ on the key parameter \Ro. However, 
 as the initial cohort size $y_0$ tends to zero, the difference between $C_2$ and twice the first raw moment $2C_1$ approaches zero.
Hence, \autoref{eq:target} is retrieved.

\subsection{Moments of time-truncated case reproductive number at some point during the outbreak}
In this section, we derive moments of truncated $C$ at an arbitrary time $j \in [0,\infty)$ during the outbreak, while accounting for the initially infectious cohort.
Up to time $j$, the expected realized infections of individuals who became infectious  at time $0 < \tau < j$ and remain infectious at time $j$ is the equal to that of the individuals who became infectious at $\tau$ and recovered at time $j$.
To capture this observation, we introduce a modified infectious period density function, denoted $g_j(t)$.
This function equals the true infectious period density  function  $f(t)$  for $t < j$, and assigns the remaining probability mass to  $t = j$. Hence, its cumulative distribution function reaches $1$ at $j$, i.e.,
\begin{equation*}
	g_j(t) = \begin{cases}
		f(t), &\text{if }t<j, \\
		F(j)\delta(t-j), & \text{otherwise.}
	\end{cases}
\end{equation*}
Accordingly, we
 redefine the size of the cohort that became infectious at time $\tau$ and recovered at time $\rho$ as follows
\begin{equation*}
	w_j(\tau, \rho) = i(\tau) g_j(\rho-\tau), \quad \tau \leq \rho \leq j. 
\end{equation*}
Note that $\int_t^j g_j(s)\;ds$ is equal to $F(t)$ for $t\leq j$.
It can be shown that the total weight, $C_0(j)$ equals 
$$
\int_{0^{-}}^{j} dt\;i(t).
$$ 
The first raw moment $C_1(j)$ is given by
\begin{align*}
	C_1(j) &= \int_{0^-}^j d\tau \int_\tau^j d\rho \; w_j(\tau,\rho) C(\tau,\rho)\\
	 &= \Ro\int_{0^-}^j d\tau \int_\tau^j d\rho i(\tau) g_j(\rho-\tau) \int_\tau^{\rho} dt x(t)\\
	& =\Ro\int_{0^-}^j d\tau \int_\tau^j dt i(\tau) x(t) \int_t^j d\rho g_j(\rho-\tau)\\
	&=  \Ro\int_{0^-}^j d\tau \int_\tau^j dt i(\tau)x(t) F(t-\tau)\\
	&=  \Ro\int_0^j dt x(t) \int_{0^-}^t d\tau i(\tau)F(t-\tau)\\
	&= \Ro\int_{0^-}^jdt\; x(t) y(t)\\
	&=\int_{0^+}^j dt\; i(t).
\end{align*}
Now we move on to calculate  $C_2(j)$.
We have
\begin{align*}
	C_2(j) &= \int_{0^-}^j d\tau \int_\tau^j d\rho \; w_j(\tau,\rho) \left(C(\tau,\rho)\right)^2\\
	&= \Ro^2 \int_{0^-}^j d\tau \int_\tau^j d\rho \;i(\tau) g_j(\rho-\tau) \left(\int_\tau^{\rho} dt x(t)\right)^2 \\
	& = 2\Ro^2 \int_{0^-}^j ds\;x(s)\int_{0^-}^s dt \; x(t) \int_{0^-}^t d\tau\;i(\tau)F(s-t)F(t-\tau)\\
	& = 2\Ro^2 \int_{0^-}^j ds\;x(s)\int_{0^-}^s dt \; x(t) F(s-t)y(t)\\
	& = 2\Ro^2 \int_{0^-}^j ds\;x(s) \int_0^s dt\; x(t)y(t) F(s-t) \\
	& = 2\Ro \int_{0^-}^j ds\;x(s) \int_0^s dt\; i(t)F(s-t)
\\
 &= 2\Ro \int_{0^-}^j ds\;x(s) \left( y(s) - y_0F(s) \right) \\
	& = 2\Ro \int_{0^-}^j ds\;x(s)y(s) -2\Ro \int_{0^-}^j ds\;x(s) y_0F(s) \\
	& = 2 \int_{0^+}^j i(s)\;ds -2\Ro \int_{0}^j ds\;x(s) y_0F(s). 
\end{align*}
Here we notice a couple of points. The second raw moment is $\emph{dependent}$ on the key parameter \Ro. However, 
as the initial cohort size $y_0$ tends to zero, the difference between $C_2$ and twice the first raw moment $2C_1$ approaches zero.
Apart from  taking the limit of $C_2$ as the initial infectious cohort size tends to zero,  increasing the truncation time increases the size of the total infected population increases, thereby diminishing the deviation introduced by $y_0$.
Hence, \autoref{eq:target} is retrieved.
